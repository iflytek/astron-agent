# Local ZWPPT MCP Service Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build, deploy, and exercise an internal SSE MCP service that exposes the six Zhiwen PPT tools and lets the imported Workflow Agent complete a real PPT generation run through Pi.

**Architecture:** A small Python package under `core/plugin/ppt` owns credential resolution, signed Zhiwen HTTP calls, and a FastMCP 1.6 SSE facade. Docker Compose runs it as `astron-agent-plugin-tool-ppt` on the existing internal network; credentials come from explicit environment variables or the existing platform-account Redis record.

**Tech Stack:** Python 3.11, uv, pytest, requests, requests-toolbelt, Redis, MCP Python SDK 1.6, Docker Compose, GitHub Actions, Playwright CLI.

## Global Constraints

- Expose exactly `/ppt/sse` and `/ppt/messages/` on internal port `3000`.
- Keep all PPT MCP ports internal; do not add a host port or public Nginx/Ingress route.
- Resolve credentials from `AIPPT_APP_ID` and `AIPPT_API_SECRET` first, then Redis key `platform_account_text:iflytek_open_platform` in `REDIS_DATABASE_CONSOLE` (default `1`).
- Never commit, return, or log complete credential values.
- Retain MIT attribution for upstream commit `d6fd686b4b6d30af064671163d1425f3d3638946`.
- Preserve the six upstream tool names and their Zhiwen API behavior.
- Do not modify the frontend or public workflow YAML schema.
- Run only one real three-page PPT generation after all non-billable checks pass.

Implement `credentials.py` with this behavior:

```python
@dataclass(frozen=True)
class Credentials:
    app_id: str
    api_secret: str


class CredentialError(RuntimeError):
    """Raised when PPT credentials cannot be resolved safely."""


def load_credentials(
    environ: Mapping[str, str] | None = None,
    redis_factory: Callable[..., RedisLike] = redis.Redis,
) -> Credentials:
    values = os.environ if environ is None else environ
    app_id = values.get("AIPPT_APP_ID", "").strip()
    api_secret = values.get("AIPPT_API_SECRET", "").strip()
    if app_id and api_secret:
        return Credentials(app_id, api_secret)

    host, port = _redis_address(values.get("REDIS_ADDR", "redis:6379"))
    client = redis_factory(
        host=host,
        port=port,
        password=values.get("REDIS_PASSWORD") or None,
        db=int(values.get("REDIS_DATABASE_CONSOLE", "1")),
        decode_responses=True,
    )
    raw = client.get("platform_account_text:iflytek_open_platform")
    managed = _managed_credentials(raw)
    if managed is None:
        raise CredentialError("PPT credentials are not configured")
    return managed
```

Convert malformed addresses, invalid JSON, Redis connection failures, incomplete values, and invalid database numbers into the same secret-safe `CredentialError`.

```python
def get_theme_list(
    self,
    pay_type: str = "not_free",
    style: str | None = None,
    color: str | None = None,
    industry: str | None = None,
    page_num: int = 2,
    page_size: int = 10,
) -> dict[str, Any]:
    params: dict[str, str | int] = {
        "payType": pay_type,
        "pageNum": page_num,
        "pageSize": page_size,
    }
    params.update(
        {
            key: value
            for key, value in {
                "style": style,
                "color": color,
                "industry": industry,
            }.items()
            if value
        }
    )
    return self._request_json(
        "GET", "/api/ppt/v2/template/list", params=params
    )

def create_ppt_task(
    self,
    text: str,
    template_id: str,
    author: str = "XXXX",
    is_card_note: bool = True,
    search: bool = False,
    is_figure: bool = True,
    ai_image: str = "normal",
) -> dict[str, Any]:
    form = MultipartEncoder(
        fields={
            "query": text,
            "templateId": template_id,
            "author": author,
            "isCardNote": str(is_card_note),
            "search": str(search),
            "isFigure": str(is_figure),
            "aiImage": ai_image,
        }
    )
    response = self._request_json(
        "POST", "/api/ppt/v2/create", data=form,
        content_type=form.content_type,
    )
    return {"sid": self._required_sid(response)}

def get_task_progress(self, sid: str) -> dict[str, Any]:
    return self._request_json(
        "GET", "/api/ppt/v2/progress", params={"sid": sid}
    )

def create_outline(
    self, text: str, language: str = "cn", search: bool = False
) -> dict[str, Any]:
    form = MultipartEncoder(
        fields={
            "query": text,
            "language": language,
            "search": str(search),
        }
    )
    return self._request_json(
        "POST", "/api/ppt/v2/createOutline", data=form,
        content_type=form.content_type,
    )

def create_outline_by_doc(
    self,
    file_name: str,
    text: str,
    file_url: str | None = None,
    file_path: str | None = None,
    language: str = "cn",
    search: bool = False,
) -> dict[str, Any]:
    if bool(file_url) == bool(file_path):
        raise ValueError("exactly one of file_url or file_path is required")
    with ExitStack() as stack:
        fields: dict[str, Any] = {
            "fileName": file_name,
            "query": text,
            "language": language,
            "search": str(search),
        }
        if file_url:
            fields["fileUrl"] = file_url
        else:
            handle = stack.enter_context(open(cast(str, file_path), "rb"))
            fields["file"] = (file_name, handle, "application/octet-stream")
        form = MultipartEncoder(fields=fields)
        return self._request_json(
            "POST", "/api/ppt/v2/createOutlineByDoc", data=form,
            content_type=form.content_type,
        )

def create_ppt_by_outline(
    self,
    text: str,
    outline: dict[str, Any] | str,
    template_id: str,
    author: str = "XXXX",
    is_card_note: bool = True,
    search: bool = False,
    is_figure: bool = True,
    ai_image: str = "normal",
) -> dict[str, Any]:
    parsed_outline = json.loads(outline) if isinstance(outline, str) else outline
    response = self._request_json(
        "POST",
        "/api/ppt/v2/createPptByOutline",
        json_body={
            "query": text,
            "outline": parsed_outline,
            "templateId": template_id,
            "author": author,
            "isCardNote": is_card_note,
            "search": search,
            "isFigure": is_figure,
            "aiImage": ai_image,
        },
    )
    return {"sid": self._required_sid(response)}
```

Use `MultipartEncoder` for `create_ppt_task`, `create_outline`, and
`create_outline_by_doc`; close local document file handles with `ExitStack`.
Send `create_ppt_by_outline` as JSON. In `_request_json`, call
`self.headers(content_type)` immediately before each request so the timestamp
and signature are fresh; pass only non-`None` request arguments to
`requests.Session.request`. `_required_sid` must validate that
`response["data"]["sid"]` is a nonempty string instead of leaking a `KeyError`.
